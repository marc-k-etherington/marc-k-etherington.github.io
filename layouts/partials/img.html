{{/* Responsive, lazy-loaded image. Usage: {{ partial "img.html" (dict "src" "/images/lab.jpg" "alt" "..." "class" "..." "style" "..." "sizes" "..." "eager" false "page" .) }} If the image lives in assets/ (or in the page's own bundle folder), Hugo builds WebP copies at several widths and the browser downloads only the size it needs. If it still lives in static/, it falls back to a plain lazy-loaded , so you can migrate images gradually without breaking anything. */}} {{ $src := .src }} {{ $alt := .alt | default "" }} {{ $class := .class | default "" }} {{ $sizes := .sizes | default "(max-width: 768px) 100vw, 800px" }} {{ $eager := .eager | default false }} {{ $style := .style | default "" }} {{ $img := false }} {{ with .page }}{{ with .Resources.Get $src }}{{ $img = . }}{{ end }}{{ end }} {{ if not $img }}{{ with resources.Get (strings.TrimPrefix "/" $src) }}{{ $img = . }}{{ end }}{{ end }} {{ if and $img (in (slice "image/jpeg" "image/png" "image/webp") $img.MediaType.Type) }} {{/* Phone photos are often stored sideways with a rotation note; apply it before resizing */}} {{ if eq $img.MediaType.Type "image/jpeg" }}{{ $img = $img.Filter images.AutoOrient }}{{ end }} {{ $srcset := slice }} {{ range slice 400 800 1200 1600 }} {{ if le . $img.Width }} {{ $r := $img.Resize (printf "%dx webp q75" .) }} {{ $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) }} {{ end }} {{ end }} {{ $w := cond (gt $img.Width 1200) 1200 $img.Width }} {{ $fallback := $img.Resize (printf "%dx webp q75" $w) }} {{ else }} {{ end }}
